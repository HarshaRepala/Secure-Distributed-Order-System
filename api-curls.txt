############################
# AUTH SERVICE
############################

# Register USER
curl -X POST "http://mymicroservices.duckdns.org/auth/v1/register" \
  -H "Content-Type: application/json" \
  -d '{
    "email":"user@example.com",
    "password":"Password@123"
  }'

# Register ADMIN
curl -X POST "http://mymicroservices.duckdns.org/auth/v1/register" \
  -H "Content-Type: application/json" \
  -d '{
    "email":"admin@example.com",
    "password":"Admin@123",
    "role":"ADMIN",
    "adminSecret":"<ADMIN_REGISTRATION_SECRET>"
  }'

# Login USER
curl -X POST "http://mymicroservices.duckdns.org/auth/v1/login" \
  -H "Content-Type: application/json" \
  -d '{
    "email":"user@example.com",
    "password":"Password@123"
  }'

# Refresh Token
curl -X POST "http://mymicroservices.duckdns.org/auth/v1/refresh" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken":"<REFRESH_TOKEN>"
  }'

# Logout
curl -X POST "http://mymicroservices.duckdns.org/auth/v1/logout" \
  -H "Content-Type: application/json" \
  -d '{
    "refreshToken":"<REFRESH_TOKEN>"
  }'

# Health Check
curl "http://mymicroservices.duckdns.org/auth/actuator/health"

# Info
curl "http://mymicroservices.duckdns.org/auth/actuator/info"

############################
# ORDER SERVICE
############################

# Create Order
curl -X POST "http://mymicroservices.duckdns.org/order/v1/orders" \
  -H "Authorization: Bearer <ACCESS_TOKEN>" \
  -H "Content-Type: application/json" \
  -d '{
    "productId":1,
    "quantity":2
  }'

# Get Orders
curl -X GET "http://mymicroservices.duckdns.org/order/v1/orders" \
  -H "Authorization: Bearer <ACCESS_TOKEN>"

# Cancel Order
curl -X POST "http://mymicroservices.duckdns.org/order/v1/orders/<ORDER_ID>/cancel" \
  -H "Authorization: Bearer <ADMIN_ACCESS_TOKEN>"

# Health Check
curl "http://mymicroservices.duckdns.org/order/actuator/health"

# Info
curl "http://mymicroservices.duckdns.org/order/actuator/info"

  -H "Content-Type: application/json" \

    "productId":1,
    "quantity":2
  }'

# Release Stock
curl -X POST "http://mymicroservices.duckdns.org/inventory/v1/release" \
  -H "Authorization: Bearer <ACCESS_TOKEN>" \
  -H "Content-Type: application/json" \
  -d '{

  }'
